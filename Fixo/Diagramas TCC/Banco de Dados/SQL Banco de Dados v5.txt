CREATE TABLE IF NOT EXISTS Documentos (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100) UNIQUE NOT NULL, 
    fk_Organizacoes_Id INTEGER(8),
    fk_Nucleos_Institucionais_Id INTEGER(8),
    fk_Curso_Id INTEGER(8),
    fk_Localizacao_Id INTEGER(8),
    fk_Integrantes_Id INTEGER(8),
    Dia DATETIME,
    Hora VARCHAR(20),
    Prefacio MEDIUMTEXT,
    Introducao MEDIUMTEXT,
    Assunto MEDIUMTEXT,
    Encerramento MEDIUMTEXT
);

CREATE TABLE IF NOT EXISTS Organizacoes (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100),
    fk_Nucleos_Institucionais_Id INTEGER(8),
    fk_Curso_Id INTEGER(8),
    fk_Local_Id INTEGER(8)
);

CREATE TABLE IF NOT EXISTS Nucleos_Institucionais (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100),
    fk_Integrantes_Id INTEGER(8)
);

CREATE TABLE IF NOT EXISTS Curso (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100)
);

CREATE TABLE IF NOT EXISTS Integrantes (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100),
    fk_Cargo_Id INTEGER(8)
);

CREATE TABLE IF NOT EXISTS cargo (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nomes VARCHAR(100),
    Sigra VARCHAR(8)
);

CREATE TABLE IF NOT EXISTS Lista_de_Integrantes (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100)
);

CREATE TABLE IF NOT EXISTS Predefinicoes (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100), 
    fk_Organizacoes_Id INTEGER(8),
    fk_Nucleos_Institucionais_Id INTEGER(8),
    fk_Curso_Id INTEGER(8),
    fk_Localizacao_Id INTEGER(8),
    fk_Integrantes_Id INTEGER(8),
    Dia DATETIME,
    Hora VARCHAR(20),
    Prefacio MEDIUMTEXT,
    Introducao MEDIUMTEXT,
    Assunto MEDIUMTEXT,
    Encerramento MEDIUMTEXT
);

CREATE TABLE IF NOT EXISTS Localizacao (
    Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    Nome VARCHAR(100)
);

CREATE TABLE IF NOT EXISTS Documentos_Organizacoes_Predefinicoes (
	Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    fk_Documentos_Id INTEGER(8) UNIQUE,
	fk_Organizacoes_Id INTEGER(8) UNIQUE,
	fk_Predefinicoes_Id INTEGER(8) UNIQUE 
);

CREATE TABLE IF NOT EXISTS Nucleo_Integrante (
	Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    fk_Nucleos_Institucionais_Id INTEGER(8) UNIQUE,
	fk_Integrantes_Id INTEGER(8) UNIQUE
);

CREATE TABLE IF NOT EXISTS Curso_Integrantes (
	Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    fk_Curso_Id INTEGER(8) UNIQUE,
	fk_Integrantes_Id INTEGER(8) UNIQUE
);

CREATE TABLE IF NOT EXISTS Integrante_lista_Integrante (
	Id INTEGER(8) PRIMARY KEY AUTO_INCREMENT,
    fk_Lista_de_Integrantes_Id INTEGER(8) UNIQUE,
	fk_Integrantes_Id INTEGER(8) UNIQUE
);
